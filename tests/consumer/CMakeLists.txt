cmake_minimum_required(VERSION 3.21)
project(arc_consumer LANGUAGES C CXX)
if(ARC_SOURCE_DIR)
    add_subdirectory("${ARC_SOURCE_DIR}" arc-build)
else()
    find_package(arc CONFIG REQUIRED)
endif()
enable_testing()
foreach(language IN ITEMS c cpp)
    configure_file(main.c "${CMAKE_CURRENT_BINARY_DIR}/main.${language}" COPYONLY)
    foreach(order IN ITEMS windows_first arc_first)
        set(name consumer_${language}_${order})
        add_executable(${name} "${CMAKE_CURRENT_BINARY_DIR}/main.${language}")
        target_link_libraries(${name} PRIVATE arc::arc)
        if(WIN32)
            target_link_libraries(${name} PRIVATE gdi32)
        endif()
        if(order STREQUAL arc_first)
            target_compile_definitions(${name} PRIVATE ARC_HEADER_FIRST)
        endif()
        file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${name}-run")
        add_test(NAME ${name} COMMAND ${name})
        set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${name}-run")
    endforeach()
endforeach()
