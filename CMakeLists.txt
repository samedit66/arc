if(NOT ARC_RELEASE_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "VERSION must contain one MAJOR.MINOR.PATCH version")
endif()

include(CMakePackageConfigHelpers)

option(ARC_DEVELOPER_WARNINGS "Enable strict warnings for arc sources" OFF)
option(ARC_BUILD_API_TESTS "Build public API tests against this library" OFF)
if(NOT DEFINED ARC_VERSION)
    set(ARC_VERSION "${PROJECT_VERSION}")
endif()
find_package(LibArchive REQUIRED)
if(WIN32)
    set(arc_platform windows)
else()
    set(arc_platform unix)
endif()
add_library(arc STATIC
    arc.c internal/writer.c internal/reader.c internal/unpack.c
    internal/libarchive_adapter.c internal/archive_paths.c internal/support.c
    internal/platform_text_${arc_platform}.c
    filekit/support.c filekit/file_system_${arc_platform}.c)
if(WIN32)
    target_sources(arc PRIVATE filekit/windows_text.c)
endif()
add_library(arc::arc ALIAS arc)
set_target_properties(arc PROPERTIES POSITION_INDEPENDENT_CODE ON C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_compile_definitions(arc PRIVATE _FILE_OFFSET_BITS=64)
target_include_directories(arc PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(arc PRIVATE LibArchive::LibArchive)
if(ARC_DEVELOPER_WARNINGS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(arc PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(ARC_BUILD_CLI)
    if(NOT ARC_VERSION MATCHES "^[A-Za-z0-9][A-Za-z0-9.+_-]*$")
        message(FATAL_ERROR "Invalid ARC_VERSION")
    endif()
    add_executable(arc_cli main.c)
    set_target_properties(arc_cli PROPERTIES OUTPUT_NAME arc C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    target_compile_definitions(arc_cli PRIVATE _FILE_OFFSET_BITS=64 ARC_VERSION="${ARC_VERSION}")
    target_link_libraries(arc_cli PRIVATE arc::arc)
    if(MINGW)
        target_link_options(arc_cli PRIVATE -municode)
    endif()
endif()
if(ARC_BUILD_API_TESTS)
    enable_testing()
    set(arc_api_tests api api_formats)
    if(WIN32)
        list(APPEND arc_api_tests api_windows_cleanup)
    endif()
    foreach(test IN LISTS arc_api_tests)
        add_executable(arc_test_${test} tests/${test}.c)
        set_target_properties(arc_test_${test} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
        target_link_libraries(arc_test_${test} PRIVATE arc::arc LibArchive::LibArchive)
        if(MSVC)
            target_compile_options(arc_test_${test} PRIVATE /UNDEBUG)
        else()
            target_compile_options(arc_test_${test} PRIVATE -UNDEBUG)
        endif()
        add_test(NAME ${test} COMMAND ${CMAKE_COMMAND}
            "-DARC_API_EXECUTABLE=$<TARGET_FILE:arc_test_${test}>"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/scripts/run-api-test.cmake")
    endforeach()
endif()

# Compute paths relative to the pc file so moving the install tree is supported.
function(arc_pc_escape input output)
    string(REPLACE "\\" "\\\\" escaped "${input}")
    string(REPLACE " " "\\ " escaped "${escaped}")
    string(REPLACE "\t" "\\\t" escaped "${escaped}")
    string(REPLACE "\"" "\\\"" escaped "${escaped}")
    string(REPLACE "'" "\\'" escaped "${escaped}")
    set(${output} "${escaped}" PARENT_SCOPE)
endfunction()
file(RELATIVE_PATH ARC_PC_PREFIX "${CMAKE_INSTALL_FULL_LIBDIR}/pkgconfig" "${CMAKE_INSTALL_PREFIX}")
if(IS_ABSOLUTE "${ARC_PC_PREFIX}")
    arc_pc_escape("${CMAKE_INSTALL_PREFIX}" ARC_PC_PREFIX_VALUE)
else()
    arc_pc_escape("${ARC_PC_PREFIX}" ARC_PC_PREFIX)
    # Quote only the dynamic pcfiledir. pkgconf already escapes that value;
    # quoting entire expanded flags would escape those spaces twice.
    set(ARC_PC_PREFIX_VALUE "\"\${pcfiledir}\"/${ARC_PC_PREFIX}")
endif()
foreach(kind LIBDIR INCLUDEDIR)
    arc_pc_escape("${CMAKE_INSTALL_${kind}}" escaped_dir)
    if(IS_ABSOLUTE "${CMAKE_INSTALL_${kind}}")
        set(ARC_PC_${kind} "${escaped_dir}")
    else()
        set(ARC_PC_${kind} "\${prefix}/${escaped_dir}")
    endif()
endforeach()
configure_file(cmake/arc.pc.in arc.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/arc.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
