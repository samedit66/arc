cmake_minimum_required(VERSION 3.21)
project(arc LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(ARC_BUILD_CLI "Build the arc command-line application" ${PROJECT_IS_TOP_LEVEL})
option(ARC_DEVELOPER_WARNINGS "Enable strict warnings for arc sources" OFF)
set(ARC_VERSION "dev" CACHE STRING "CLI version")
find_package(LibArchive REQUIRED)
if(WIN32)
    set(arc_platform windows)
else()
    set(arc_platform unix)
endif()
add_library(arc STATIC
    arc.c internal/writer.c internal/reader.c internal/unpack.c
    internal/libarchive_adapter.c internal/archive_paths.c internal/support.c
    internal/platform_text_${arc_platform}.c
    filekit/support.c filekit/file_system_${arc_platform}.c)
if(WIN32)
    target_sources(arc PRIVATE filekit/windows_text.c)
endif()
add_library(arc::arc ALIAS arc)
set_target_properties(arc PROPERTIES POSITION_INDEPENDENT_CODE ON C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_compile_definitions(arc PRIVATE _FILE_OFFSET_BITS=64)
target_include_directories(arc PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(arc PRIVATE LibArchive::LibArchive)
if(ARC_DEVELOPER_WARNINGS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(arc PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(ARC_BUILD_CLI)
    if(NOT ARC_VERSION MATCHES "^[A-Za-z0-9][A-Za-z0-9.+_-]*$")
        message(FATAL_ERROR "Invalid ARC_VERSION")
    endif()
    add_executable(arc_cli main.c)
    set_target_properties(arc_cli PROPERTIES OUTPUT_NAME arc C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    target_compile_definitions(arc_cli PRIVATE _FILE_OFFSET_BITS=64 ARC_VERSION="${ARC_VERSION}")
    target_link_libraries(arc_cli PRIVATE arc::arc)
    if(MINGW)
        target_link_options(arc_cli PRIVATE -municode)
    endif()
endif()
install(TARGETS arc EXPORT arcTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES arc.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT arcTargets NAMESPACE arc:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/arc)
configure_package_config_file(cmake/arcConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/arcConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/arc)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/arcConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/arc)
# Compute paths relative to the pc file so moving the install tree is supported.
file(RELATIVE_PATH ARC_PC_PREFIX "/${CMAKE_INSTALL_LIBDIR}/pkgconfig" "/")
configure_file(cmake/arc.pc.in arc.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/arc.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
