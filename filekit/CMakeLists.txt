cmake_minimum_required(VERSION 3.21)
project(filekit LANGUAGES C)
option(FILEKIT_SANITIZE "Enable ASan/UBSan" OFF)
include(sources.cmake)
add_library(filekit STATIC ${filekit_sources})
add_executable(filekit_check tests.c)
target_link_libraries(filekit_check PRIVATE filekit)
foreach(target filekit filekit_check)
    set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    target_compile_definitions(${target} PRIVATE _FILE_OFFSET_BITS=64)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wstrict-prototypes -Wmissing-prototypes)
    endif()
    if(FILEKIT_SANITIZE)
        if(WIN32 OR NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
            message(FATAL_ERROR "FILEKIT_SANITIZE requires GCC/Clang on macOS or Linux")
        endif()
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
if(MSVC)
    target_compile_options(filekit_check PRIVATE /UNDEBUG)
else()
    target_compile_options(filekit_check PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME filekit COMMAND ${CMAKE_COMMAND}
    "-DARC_API_EXECUTABLE=$<TARGET_FILE:filekit_check>"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/run-test.cmake")
